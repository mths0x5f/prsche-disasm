cmake_minimum_required(VERSION 3.24)
project(formatter_output_166 LANGUAGES CXX)
if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Run166 requires MSVC Win32/x86")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(formatter_output_probe "${ROOT}/source/recovered/formatter_output_probe.cpp" "${ROOT}/source/recovered/Porsche.exe/formatter_output.cpp" "${ROOT}/source/recovered/Porsche.exe/formatter_cleanup.cpp")
target_include_directories(formatter_output_probe PRIVATE "${ROOT}/source/include")
target_compile_features(formatter_output_probe PRIVATE cxx_std_17)
target_compile_options(formatter_output_probe PRIVATE /W4 /Gy /EHsc /permissive-)
target_compile_definitions(formatter_output_probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(formatter_output_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_link_options(formatter_output_probe PRIVATE /OPT:REF)
