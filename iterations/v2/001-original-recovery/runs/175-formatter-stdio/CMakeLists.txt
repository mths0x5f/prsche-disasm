cmake_minimum_required(VERSION 3.24)
project(formatter_stdio_175 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "Run175 requires MSVC Win32")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(formatter_stdio_probe "${ROOT}/source/recovered/formatter_stdio_probe.cpp" "${ROOT}/source/recovered/Porsche.exe/formatter_stdio.cpp" "${ROOT}/source/recovered/Porsche.exe/formatter_runtime_storage.cpp")
target_include_directories(formatter_stdio_probe PRIVATE "${ROOT}/source/include")
target_compile_features(formatter_stdio_probe PRIVATE cxx_std_17)
target_compile_options(formatter_stdio_probe PRIVATE /W4 /permissive- /EHsc)
target_compile_definitions(formatter_stdio_probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(formatter_stdio_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
