cmake_minimum_required(VERSION 3.20)
project(network_transport_168 LANGUAGES CXX)
if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "Run168 requires MSVC Win32")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
execute_process(COMMAND py -3 "${CMAKE_CURRENT_LIST_DIR}/extract-count.py" "${CMAKE_BINARY_DIR}/canonical_count.cpp" COMMAND_ERROR_IS_FATAL ANY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/source/recovered/Porsche.exe/network_workers_members.cpp" "${CMAKE_CURRENT_LIST_DIR}/extract-count.py")
add_executable(network_transport_probe "${ROOT}/source/recovered/network_transport_probe.cpp" "${ROOT}/source/recovered/Porsche.exe/network_transport.cpp" "${CMAKE_BINARY_DIR}/canonical_count.cpp" "${ROOT}/source/recovered/Porsche.exe/io_lists.cpp")
target_include_directories(network_transport_probe PRIVATE "${ROOT}/source/include")
target_compile_features(network_transport_probe PRIVATE cxx_std_17)
target_compile_options(network_transport_probe PRIVATE /W4 /EHsc /permissive-)
target_compile_definitions(network_transport_probe PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
set_target_properties(network_transport_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
