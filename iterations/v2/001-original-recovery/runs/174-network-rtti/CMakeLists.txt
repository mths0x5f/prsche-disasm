cmake_minimum_required(VERSION 3.20)
project(network_rtti_174 LANGUAGES CXX)
if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "Run174 requires MSVC Win32")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(rtti_probe "${ROOT}/source/recovered/rtti_probe.cpp" "${ROOT}/source/recovered/Porsche.exe/rtti.cpp")
target_include_directories(rtti_probe PRIVATE "${ROOT}/source/include")
target_compile_features(rtti_probe PRIVATE cxx_std_17)
target_compile_options(rtti_probe PRIVATE /W4 /EHsc /permissive-)
target_compile_definitions(rtti_probe PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
set_target_properties(rtti_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
