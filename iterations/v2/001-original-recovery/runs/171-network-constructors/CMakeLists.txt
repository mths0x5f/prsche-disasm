cmake_minimum_required(VERSION 3.20)
project(network_constructors_171 LANGUAGES CXX)
if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "Run171 requires MSVC Win32")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(network_constructors_probe "${ROOT}/source/recovered/network_constructors_probe.cpp" "${ROOT}/source/recovered/Porsche.exe/network_constructors.cpp")
target_include_directories(network_constructors_probe PRIVATE "${ROOT}/source/include")
target_compile_features(network_constructors_probe PRIVATE cxx_std_17)
target_compile_options(network_constructors_probe PRIVATE /W4 /EHsc /permissive-)
target_compile_definitions(network_constructors_probe PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
set_target_properties(network_constructors_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
