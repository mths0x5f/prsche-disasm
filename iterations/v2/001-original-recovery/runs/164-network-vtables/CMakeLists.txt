cmake_minimum_required(VERSION 3.20)
project(network_vtables_164 LANGUAGES CXX)
if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "Run164 requires MSVC Win32")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
execute_process(COMMAND py -3 "${CMAKE_CURRENT_LIST_DIR}/extract-helpers.py" "${CMAKE_BINARY_DIR}/canonical_helpers.cpp" COMMAND_ERROR_IS_FATAL ANY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/source/recovered/Porsche.exe/network_workers_members.cpp" "${ROOT}/source/recovered/Porsche.exe/network_cleanup.cpp" "${CMAKE_CURRENT_LIST_DIR}/extract-helpers.py")
add_executable(network_vtables_probe "${ROOT}/source/recovered/network_vtables_probe.cpp" "${ROOT}/source/recovered/Porsche.exe/network_vtables.cpp" "${CMAKE_BINARY_DIR}/canonical_helpers.cpp")
target_include_directories(network_vtables_probe PRIVATE "${ROOT}/source/include")
target_compile_features(network_vtables_probe PRIVATE cxx_std_17)
target_compile_options(network_vtables_probe PRIVATE /W4 /EHsc /permissive- /Gy /Gw)
target_compile_definitions(network_vtables_probe PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_link_options(network_vtables_probe PRIVATE /OPT:REF)
set_target_properties(network_vtables_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
