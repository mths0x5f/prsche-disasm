cmake_minimum_required(VERSION 3.20)
project(network_workers_members_159 LANGUAGES CXX)
if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Run159 requires native MSVC Win32/x86")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(network_workers_members_probe "${ROOT}/source/recovered/network_workers_members_probe.cpp")
foreach(NAME network_workers_members network_session network_services network_cleanup network_lifecycle context_effects io_lists io_worker_lists file_events)
  target_sources(network_workers_members_probe PRIVATE "${ROOT}/source/recovered/Porsche.exe/${NAME}.cpp")
endforeach()
target_include_directories(network_workers_members_probe PRIVATE "${ROOT}/source/include")
target_compile_features(network_workers_members_probe PRIVATE cxx_std_17)
target_compile_definitions(network_workers_members_probe PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(network_workers_members_probe PRIVATE /W4 /EHsc /permissive-)
set_target_properties(network_workers_members_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
